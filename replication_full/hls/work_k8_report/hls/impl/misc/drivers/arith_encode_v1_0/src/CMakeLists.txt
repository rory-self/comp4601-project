cmake_minimum_required(VERSION 3.15)
project(arith_encode)
find_package(common)
collector_create (PROJECT_LIB_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}")
collect (PROJECT_LIB_SOURCES xarith_encode.c)
collect (PROJECT_LIB_SOURCES xarith_encode_sinit.c)
collect (PROJECT_LIB_SOURCES xarith_encode_linux.c)
collect (PROJECT_LIB_SOURCES xarith_encode_g.c)
collector_list (_sources PROJECT_LIB_SOURCES)
collector_create (PROJECT_LIB_HEADERS "${CMAKE_CURRENT_SOURCE_DIR}")
collect (PROJECT_LIB_HEADERS xarith_encode.h)
collect (PROJECT_LIB_HEADERS xarith_encode_hw.h)
collector_list (_headers PROJECT_LIB_HEADERS)
file(COPY ${_headers} DESTINATION ${CMAKE_BINARY_DIR}/include)
include_directories(${CMAKE_BINARY_DIR}/include)
add_library(arith_encode STATIC ${_sources})
set_target_properties(arith_encode PROPERTIES LINKER_LANGUAGE C)
